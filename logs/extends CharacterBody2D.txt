extends CharacterBody2D

# This sets how fast your player moves
var speed = 400 

func _physics_process(delta):
    # 1. Get the direction from player input
    # "ui_left", "ui_right", etc. are default keys (Arrows/WASD)
    var direction = Input.get_vector("ui_left", "ui_right", "ui_up", "ui_down")
    
    # 2. Apply the direction to our velocity (speed)
    velocity = direction * speed
    
    # 3. Move the character and handle sliding against walls
    move_and_slide()
